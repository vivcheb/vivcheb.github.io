Arduino 
// Pin Definitions (RAMPS 1.4 / MKS Base)
const int stepPins[3] = {A0, A6, 46};  // X, Y, Z Step pins
const int dirPins[3]  = {A1, A7, 48};  // X, Y, Z Direction pins
const int enPins[3]   = {38, A2, A8};  // X, Y, Z Enable pins

int d = 250; // Pulse delay in microseconds (adjust speed)

void setup() {
  Serial.begin(115200);

  for (int i = 0; i < 3; i++) {
    pinMode(stepPins[i], OUTPUT);
    pinMode(dirPins[i], OUTPUT);
    pinMode(enPins[i], OUTPUT);
    digitalWrite(enPins[i], HIGH); // HIGH = Disabled (standby)
  }

  Serial.println("Individual Motor Controller Ready.");
  Serial.println("Command format: <Motor> <Steps>");
  Serial.println("Examples: 'X 25000', 'Y 25000', 'Z 25000'");
}

// Drive a single motor directly
void runSingleMotor(int motorIndex, long steps) {
  if (motorIndex < 0 || motorIndex > 2) return;

  // Determine direction
  bool forward = (steps >= 0);
  long stepCount = abs(steps);

  // Enable target motor and set direction
  digitalWrite(enPins[motorIndex], LOW); // LOW = Enabled
  digitalWrite(dirPins[motorIndex], forward ? HIGH : LOW);
  delayMicroseconds(5);

  // Step pulse loop
  for (long ct = 0; ct < stepCount; ct++) {
    digitalWrite(stepPins[motorIndex], HIGH);
    delayMicroseconds(d);
    digitalWrite(stepPins[motorIndex], LOW);
    delayMicroseconds(d);
  }

  // Turn driver off to prevent heating
  digitalWrite(enPins[motorIndex], HIGH);
}

void loop() {
  if (Serial.available() > 0) {
    char motorChoice = Serial.read();

    // Check if the character is X, Y, or Z (case-insensitive)
    int motorIdx = -1;
    if (motorChoice == 'X' || motorChoice == 'x') motorIdx = 0;
    else if (motorChoice == 'Y' || motorChoice == 'y') motorIdx = 1;
    else if (motorChoice == 'Z' || motorChoice == 'z') motorIdx = 2;

    if (motorIdx != -1) {
      long steps = Serial.parseInt(); // Reads step count and sign (+ or -)

      // Clear remaining newline/carriage returns
      while (Serial.available() > 0) {
        Serial.read();
      }

      Serial.print("Running Motor ");
      Serial.print(motorChoice);
      Serial.print(" for ");
      Serial.print(steps);
      Serial.println(" steps.");

      runSingleMotor(motorIdx, steps);
      Serial.println("Done.");
    }
  }
}
